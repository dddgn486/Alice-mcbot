   private static void syncBotInventorySlots(ServerPlayer bot) {
       for (int i = 0; i < bot.getInventory().getContainerSize(); i++) {
           bot.connection.send(new ClientboundContainerSetSlotPacket(
               -2, 0, i, bot.getInventory().getItem(i)
           ));
